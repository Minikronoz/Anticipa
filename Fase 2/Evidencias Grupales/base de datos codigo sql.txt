-- ── Tabla: COLEGIO ──────────────────────────────────────────
CREATE TABLE colegio (
    id_colegio         NUMBER(10) GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nombre             VARCHAR2(100) NOT NULL,
    rut                VARCHAR2(20) UNIQUE,
    direccion          VARCHAR2(255),
    telefono           VARCHAR2(20),
    correo             VARCHAR2(100),
    contacto_nombre    VARCHAR2(100),
    activo             NUMBER(1) DEFAULT 1 CHECK (activo IN (0,1)),
    plan               VARCHAR2(20) DEFAULT 'basico',
    fecha_contrato     DATE,
    fecha_vencimiento  DATE
);

-- ── Tabla: ROL ──────────────────────────────────────────────
CREATE TABLE rol (
    id_rol     NUMBER(10) GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nombre_rol VARCHAR2(50) NOT NULL UNIQUE
);

-- ── Tabla: CURSO ────────────────────────────────────────────
CREATE TABLE curso (
    id_curso          NUMBER(10) GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nivel_academico   VARCHAR2(10) NOT NULL,
    letra_academica   VARCHAR2(1),
    colegio_id_colegio NUMBER(10) REFERENCES colegio(id_colegio)
);

-- ── Tabla: USUARIO ──────────────────────────────────────────
CREATE TABLE usuario (
    id_usuario         NUMBER(10) GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    rol_id_rol         NUMBER(10) NOT NULL REFERENCES rol(id_rol),
    nombre             VARCHAR2(100) NOT NULL,
    email              VARCHAR2(100) NOT NULL UNIQUE,
    password_hash      VARCHAR2(255) NOT NULL,
    fecha_registro     TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    reset_token        VARCHAR2(100) UNIQUE,
    reset_token_expiry TIMESTAMP WITH TIME ZONE,
    curso_id_curso     NUMBER(10) REFERENCES curso(id_curso),
    es_admin           NUMBER(1) DEFAULT 0 CHECK (es_admin IN (0,1))
);

-- ── Tabla: ESTUDIANTE ───────────────────────────────────────
CREATE TABLE estudiante (
    id_estudiante       NUMBER(10) GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nombre              VARCHAR2(100) NOT NULL,
    fecha_nacimiento    DATE NOT NULL,
    codigo_vinculacion  VARCHAR2(7) UNIQUE,
    puntos_totales      NUMBER(10) DEFAULT 0,
    diagnostico         VARCHAR2(50),
    estado              VARCHAR2(20),
    creado_en           TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    usuario_id_usuario  NUMBER(10) NOT NULL REFERENCES usuario(id_usuario),
    curso_id_curso      NUMBER(10) NOT NULL REFERENCES curso(id_curso)
);

-- ── Tabla: VINCULACION_HISTORIAL ────────────────────────────
CREATE TABLE vinculacion_historial (
    id_vinculo         NUMBER(10) GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    usuario_id_usuario NUMBER(10) NOT NULL REFERENCES usuario(id_usuario),
    id_estudiante      NUMBER(10) NOT NULL REFERENCES estudiante(id_estudiante),
    rol_id_rol         NUMBER(10) NOT NULL REFERENCES rol(id_rol),
    fecha_inicio       TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    fecha_termino      TIMESTAMP WITH TIME ZONE,
    motivo_cambio      VARCHAR2(255)
);

CREATE UNIQUE INDEX idx_un_vinculo_activo
ON vinculacion_historial (usuario_id_usuario, id_estudiante)
WHERE fecha_termino IS NULL;

-- ── Tabla: PICTOGRAMA ───────────────────────────────────────
CREATE TABLE pictograma (
    id_pictograma NUMBER(10) GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nombre_imagen VARCHAR2(100) NOT NULL,
    url           CLOB NOT NULL,
    categoria     VARCHAR2(50)
);

-- ── Tabla: CATALOGO_ACTIVIDAD ───────────────────────────────
CREATE TABLE catalogo_actividad (
    id_catalogo              NUMBER(10) GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nombre_predeterminado    VARCHAR2(100) NOT NULL,
    pictograma_id_pictograma NUMBER(10) REFERENCES pictograma(id_pictograma)
);

-- ── Tabla: ACTIVIDAD ────────────────────────────────────────
CREATE TABLE actividad (
    id_actividad                   NUMBER(10) GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    estudiante_id_estudiante       NUMBER(10) NOT NULL REFERENCES estudiante(id_estudiante) ON DELETE CASCADE,
    usuario_id_usuario             NUMBER(10) NOT NULL REFERENCES usuario(id_usuario),
    pictograma_id_pictograma       NUMBER(10) REFERENCES pictograma(id_pictograma),
    catalogo_actividad_id_catalogo NUMBER(10) REFERENCES catalogo_actividad(id_catalogo),
    nombre_tarea                   VARCHAR2(100) NOT NULL,
    hora_inicio                    TIMESTAMP NOT NULL,
    hora_fin                       TIMESTAMP NOT NULL,
    es_completada                  NUMBER(1) DEFAULT 0 CHECK (es_completada IN (0,1)),
    fecha_actividad                DATE NOT NULL,
    fecha_creacion                 TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- ── Tabla: CONFIGURACION_ALERTA ─────────────────────────────
CREATE TABLE configuracion_alerta (
    id_alerta              NUMBER(10) GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    actividad_id_actividad NUMBER(10) NOT NULL UNIQUE REFERENCES actividad(id_actividad) ON DELETE CASCADE,
    minutos_anticipacion   VARCHAR2(5) DEFAULT '5' CHECK (minutos_anticipacion IN ('2','5','10','15')),
    tipo_sonido            VARCHAR2(20) DEFAULT 'suave' CHECK (tipo_sonido IN ('suave','moderado','silencio')),
    parpadeo_visual        NUMBER(1) DEFAULT 1 CHECK (parpadeo_visual IN (0,1))
);

-- ── Tabla: HISTORIAL_CUMPLIMIENTO ───────────────────────────
CREATE TABLE historial_cumplimiento (
    id_log                 NUMBER(10) GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    actividad_id_actividad NUMBER(10) NOT NULL REFERENCES actividad(id_actividad),
    fecha_cumplimiento     TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    observaciones          VARCHAR2(255)
);

-- ── Tabla: RECOMPENSA_DISPONIBLE ────────────────────────────
CREATE TABLE recompensa_disponible (
    id_recompensa            NUMBER(10) GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    estudiante_id_estudiante NUMBER(10) NOT NULL REFERENCES estudiante(id_estudiante) ON DELETE CASCADE,
    nombre_recompensa        VARCHAR2(100) NOT NULL,
    recompensa_url           CLOB,
    meta_estrellas           NUMBER(10) DEFAULT 5,
    estado_logro             NUMBER(1) DEFAULT 0 CHECK (estado_logro IN (0,1)),
    fecha_logro              TIMESTAMP WITH TIME ZONE
);

-- ── Tabla: REGISTRO_ESTRELLA_DIARIA ─────────────────────────
CREATE TABLE registro_estrella_diaria (
    id_registro              NUMBER(10) GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    estudiante_id_estudiante NUMBER(10) NOT NULL REFERENCES estudiante(id_estudiante),
    fecha                    DATE DEFAULT CURRENT_DATE,
    estrellas_ganadas        NUMBER(10) DEFAULT 0 CHECK (estrellas_ganadas >= 0),
    CONSTRAINT unq_estudiante_fecha UNIQUE (estudiante_id_estudiante, fecha)
);

-- ── Tabla: ENCUESTA_DIARIA ──────────────────────────────────
CREATE TABLE encuesta_diaria (
    id_encuesta               NUMBER(10) GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    estudiante_id_estudiante  NUMBER(10) NOT NULL REFERENCES estudiante(id_estudiante) ON DELETE CASCADE,
    usuario_id_usuario        NUMBER(10) NOT NULL REFERENCES usuario(id_usuario),
    fecha                     DATE DEFAULT CURRENT_DATE,
    tuvo_desregulacion        NUMBER(1) NOT NULL CHECK (tuvo_desregulacion IN (0,1)),
    cantidad                  NUMBER(10),
    motivo                    VARCHAR2(100),
    otro_motivo               CLOB,
    observacion               CLOB,
    CONSTRAINT unq_encuesta_por_dia UNIQUE (estudiante_id_estudiante, fecha)
);
